cmake_minimum_required(VERSION 3.20)
project(NavierStokes2D VERSION 0.3.0 LANGUAGES CXX)

option(NS2D_OPENMP "Enable OpenMP and threaded FFTW" ON)
option(NS2D_MPI "Build the hybrid MPI/OpenMP backend" ON)
option(NS2D_CUDA "Build the CUDA/cuFFT backend" ON)
option(NS2D_BACKEND_TESTS "Run MPI/CUDA integration tests (requires runtime access)" OFF)
set(NS2D_CUDA_ARCHITECTURES "" CACHE STRING "Optional CUDA architecture override")

find_path(FFTW_INCLUDE_DIR fftw3.h REQUIRED)
find_library(FFTW_LIBRARY NAMES fftw3 REQUIRED)
find_library(FFTW_THREADS_LIBRARY NAMES fftw3_threads)

add_library(ns2d_common
    src/parameters.cpp
    src/fftw_utils.cpp
    src/spectral.cpp
    src/output.cpp
    src/output_transaction.cpp
    src/solver.cpp)
target_include_directories(ns2d_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src ${FFTW_INCLUDE_DIR})
target_link_libraries(ns2d_common PUBLIC ${FFTW_LIBRARY})
target_compile_features(ns2d_common PUBLIC cxx_std_20)

if(NS2D_OPENMP)
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(ns2d_common PUBLIC OpenMP::OpenMP_CXX)
  endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ns2d_common PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(navier_stokes_cpu src/main.cpp src/backend_cpu.cpp)
target_link_libraries(navier_stokes_cpu PRIVATE ns2d_common)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(navier_stokes_cpu PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FFTW_THREADS_LIBRARY)
  target_link_libraries(navier_stokes_cpu PRIVATE ${FFTW_THREADS_LIBRARY})
  target_compile_definitions(navier_stokes_cpu PRIVATE NS2D_HAVE_FFTW_THREADS=1)
else()
  message(WARNING "FFTW threads library not found; CPU FFT plans remain single-threaded")
endif()

if(NS2D_MPI)
  find_package(MPI COMPONENTS CXX)
  find_path(FFTW_MPI_INCLUDE_DIR fftw3-mpi.h)
  find_library(FFTW_MPI_LIBRARY NAMES fftw3_mpi)
  if(MPI_CXX_FOUND AND FFTW_MPI_INCLUDE_DIR AND FFTW_MPI_LIBRARY)
    add_executable(navier_stokes_mpi src/main.cpp src/backend_mpi.cpp)
    target_include_directories(navier_stokes_mpi PRIVATE ${FFTW_MPI_INCLUDE_DIR})
    target_link_libraries(navier_stokes_mpi PRIVATE ns2d_common MPI::MPI_CXX
                                                    ${FFTW_MPI_LIBRARY})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(navier_stokes_mpi PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(FFTW_THREADS_LIBRARY)
      target_link_libraries(navier_stokes_mpi PRIVATE ${FFTW_THREADS_LIBRARY})
      target_compile_definitions(navier_stokes_mpi PRIVATE NS2D_HAVE_FFTW_THREADS=1)
    endif()
  else()
    message(WARNING "MPI and/or FFTW-MPI not found; navier_stokes_mpi will not be built")
  endif()
endif()

if(NS2D_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_executable(navier_stokes_cuda src/main.cpp src/backend_cuda.cu)
    target_link_libraries(navier_stokes_cuda PRIVATE ns2d_common CUDA::cudart CUDA::cufft)
    set_target_properties(navier_stokes_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
    if(NOT NS2D_CUDA_ARCHITECTURES STREQUAL "")
      set_property(TARGET navier_stokes_cuda PROPERTY CUDA_ARCHITECTURES "${NS2D_CUDA_ARCHITECTURES}")
    endif()
  else()
    message(WARNING "CUDA toolkit not found; navier_stokes_cuda will not be built")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(ns2d_numerics_tests tests/numerics.cpp src/backend_cpu.cpp)
  target_link_libraries(ns2d_numerics_tests PRIVATE ns2d_common)
  if(FFTW_THREADS_LIBRARY)
    target_link_libraries(ns2d_numerics_tests PRIVATE ${FFTW_THREADS_LIBRARY})
    target_compile_definitions(ns2d_numerics_tests PRIVATE NS2D_HAVE_FFTW_THREADS=1)
  endif()
  add_test(NAME nonlinear_direct_dft COMMAND ns2d_numerics_tests)
  set(NS2D_SMOKE_DIR "${CMAKE_CURRENT_BINARY_DIR}/smoke")
  file(MAKE_DIRECTORY "${NS2D_SMOKE_DIR}")
  configure_file(tests/smoke.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/smoke.cmake" @ONLY)
  add_test(NAME cpu_output_smoke COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_BINARY_DIR}/smoke.cmake")
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    foreach(check IN ITEMS numerics output)
      add_test(NAME regression_${check} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/regression.py ${check}
        --cpu $<TARGET_FILE:navier_stokes_cpu>)
    endforeach()
    if(NS2D_BACKEND_TESTS)
      foreach(backend IN ITEMS mpi cuda)
        if(TARGET navier_stokes_${backend})
          set(launcher)
          if(backend STREQUAL "mpi")
            list(APPEND launcher --mpiexec ${MPIEXEC_EXECUTABLE}
              --mpi-numproc-flag=${MPIEXEC_NUMPROC_FLAG})
          endif()
          add_test(NAME regression_${backend} COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/regression.py backend
            --cpu $<TARGET_FILE:navier_stokes_cpu>
            --candidate $<TARGET_FILE:navier_stokes_${backend}> ${launcher})
          set_tests_properties(regression_${backend} PROPERTIES LABELS "backend" TIMEOUT 300)
        endif()
      endforeach()
    endif()
  else()
    message(WARNING "Python 3 not found; end-to-end regression tests will not be registered")
  endif()
endif()
